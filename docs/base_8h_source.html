591static inline int
592ccs_datum_cmp(const ccs_datum_t a, const ccs_datum_t b)
593{
594 if (a.type < b.type) {
595 return -1;
596 } else if (a.type > b.type) {
597 return 1;
598 } else {
599 switch (a.type) {
600 case CCS_DATA_TYPE_STRING:
601 if (a.value.s == b.value.s)
602 return 0;
603 else if (!a.value.s)
604 return -1;
605 else if (!b.value.s)
606 return 1;
607 else
608 return strcmp(a.value.s, b.value.s);
609 break;
610 case CCS_DATA_TYPE_INT:
611 return a.value.i < b.value.i ? -1 :
612 a.value.i > b.value.i ? 1 :
613 0;
614 break;
615 case CCS_DATA_TYPE_FLOAT:
616 return a.value.f < b.value.f ? -1 :
617 a.value.f > b.value.f ? 1 :
618 0;
619 break;
620 case CCS_DATA_TYPE_NONE:
621 case CCS_DATA_TYPE_INACTIVE:
622 return 0;
623 break;
624 default:
625 return memcmp(
626 &(a.value), &(b.value), sizeof(ccs_value_t));
627 }
628 }
